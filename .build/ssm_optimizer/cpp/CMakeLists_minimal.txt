cmake_minimum_required(VERSION 3.16)
project(SSMOptimizer VERSION 1.0.0 LANGUAGES CXX C)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Options
option(SSM_USE_SIMD "Enable SIMD optimizations (AVX/AVX2)" ON)
option(SSM_USE_OPENMP "Enable OpenMP parallelization" ON)
option(SSM_BUILD_ONNX "Build with ONNX Runtime support" OFF)

# Compiler flags
if(MSVC)
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /O2 /fp:fast /W4")
    if(SSM_USE_SIMD)
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /arch:AVX2")
        add_compile_definitions(SSM_USE_AVX2)
    endif()
else()
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O3 -ffast-math -Wall")
    if(SSM_USE_SIMD)
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -mavx2 -mfma")
        add_compile_definitions(SSM_USE_AVX2)
    endif()
endif()

# OpenMP
if(SSM_USE_OPENMP)
    find_package(OpenMP)
    if(OpenMP_CXX_FOUND)
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${OpenMP_CXX_FLAGS}")
    endif()
endif()

# Source files (minimal version without ONNX)
set(SSM_SOURCES
    src/ssm_optimizer.cpp
    src/selective_scan.cpp
    src/state_cache.cpp
)

if(SSM_BUILD_ONNX)
    list(APPEND SSM_SOURCES src/onnx_custom_op.cpp)
endif()

# Header files
set(SSM_HEADERS
    ssm_optimizer.h
    src/selective_scan.h
    src/state_cache.h
    src/simd_utils.h
)

# Create shared library
add_library(SSMOptimizer SHARED ${SSM_SOURCES} ${SSM_HEADERS})

target_compile_definitions(SSMOptimizer PRIVATE SSM_OPTIMIZER_EXPORTS)

if(SSM_USE_OPENMP AND OpenMP_CXX_FOUND)
    target_link_libraries(SSMOptimizer OpenMP::OpenMP_CXX)
endif()

# ONNX Runtime (optional)
if(SSM_BUILD_ONNX)
    if(DEFINED ENV{ONNXRUNTIME_ROOT})
        set(ONNXRUNTIME_ROOT $ENV{ONNXRUNTIME_ROOT})
    else()
        set(ONNXRUNTIME_ROOT "C:/Program Files/onnxruntime")
    endif()
    
    target_include_directories(SSMOptimizer PRIVATE
        ${ONNXRUNTIME_ROOT}/include
        ${CMAKE_CURRENT_SOURCE_DIR}
    )
    
    target_link_directories(SSMOptimizer PRIVATE
        ${ONNXRUNTIME_ROOT}/lib
    )
    
    target_link_libraries(SSMOptimizer onnxruntime)
endif()

# Install
install(TARGETS SSMOptimizer
    RUNTIME DESTINATION bin
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
)

install(FILES ssm_optimizer.h DESTINATION include)

# Print configuration
message(STATUS "")
message(STATUS "SSM Optimizer Configuration:")
message(STATUS "  Version: ${PROJECT_VERSION}")
message(STATUS "  SIMD: ${SSM_USE_SIMD}")
message(STATUS "  OpenMP: ${SSM_USE_OPENMP}")
message(STATUS "  ONNX Runtime: ${SSM_BUILD_ONNX}")
message(STATUS "")
